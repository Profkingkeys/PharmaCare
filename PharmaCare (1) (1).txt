// ── SLIDESHOW ──
const slides = document.querySelectorAll('.slide');
const dotsEl = document.getElementById('slideDots');
const captionEl = document.getElementById('slideCaption');
const progressEl = document.getElementById('slideProgress');
let current = 0;
let slideTimer, progressInterval, progressVal = 0;
const DURATION = 4500;

slides.forEach((_, i) => {
  const d = document.createElement('div');
  d.className = 'dot' + (i === 0 ? ' active' : '');
  d.onclick = () => goTo(i);
  dotsEl.appendChild(d);
});

function goTo(n) {
  slides[current].classList.remove('active');
  dotsEl.children[current].classList.remove('active');
  current = (n + slides.length) % slides.length;
  slides[current].classList.add('active');
  dotsEl.children[current].classList.add('active');
  captionEl.textContent = slides[current].dataset.caption || '';
  resetProg();
}

function changeSlide(d) { goTo(current + d); }

function resetProg() {
  progressVal = 0;
  progressEl.style.width = '0%';
  clearInterval(progressInterval);
  progressInterval = setInterval(() => {
    progressVal += 100 / (DURATION / 80);
    if (progressVal >= 100) progressVal = 100;
    progressEl.style.width = progressVal + '%';
  }, 80);
  clearTimeout(slideTimer);
  slideTimer = setTimeout(() => goTo(current + 1), DURATION);
}
resetProg();

  if (!name) { showToast('Please enter your name'); document.getElementById('pat-name').focus(); return; }
  if (!phone) { showToast('Please enter your phone number'); document.getElementById('pat-phone').focus(); return; }
  if (!address) { showToast('Please enter your delivery address'); document.getElementById('pat-address').focus(); return; }

  const drugList = drugs.map((d, i) => `  ${i+1}. ${d.name}${d.qty ? ' — Qty: ' + d.qty : ''}`).join('\n');

  const msg = `Hello PharmaCare,\n\nI would like to place a medication order.\n\n*Medications Requested:*\n${drugList}\n\n*Patient Details:*\nName: ${name}\nPhone: ${phone}\nDelivery Address: ${address}${notes ? '\n\nNotes / Prescription Info: ' + notes : ''}\n\nKindly confirm availability, pricing, and delivery timeline at your earliest convenience. Thank you.`;

  window.open('https://wa.me/' + WA + '?text=' + encodeURIComponent(msg), '_blank');
}

function matchIntent(msg) {
  const m = msg.toLowerCase();
  for (const [intent, words] of Object.entries(pharmaBot)) {
    if (words.some(w => m.includes(w))) return intent;
  }
  return null;
}

function timeNow() { return new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }); }

function addMsg(text, role, withWA = false, waQ = '') {
  const c = document.getElementById('chatMessages');
  const el = document.createElement('div');
  el.className = 'msg ' + role;
  let html = `<div class="msg-bubble">${text}</div>`;
  if (withWA) {
    const enc = encodeURIComponent('Hello PharmaCare, I need assistance with: ' + waQ);
    html += `<a class="chat-wa-link" href="https://wa.me/${WA}?text=${enc}" target="_blank">
      <i class="fa-brands fa-whatsapp"></i> Contact Pharmacist
    </a>`;
  }
  html += `<div class="msg-time">${timeNow()}</div>`;
  el.innerHTML = html;
  c.appendChild(el);
  c.scrollTop = c.scrollHeight;
}

function showTyping() {
  const c = document.getElementById('chatMessages');
  const el = document.createElement('div');
  el.className = 'msg bot'; el.id = 'typingInd';
  el.innerHTML = `<div class="typing-indicator"><div class="typing-dot"></div><div class="typing-dot"></div><div class="typing-dot"></div></div>`;
  c.appendChild(el); c.scrollTop = c.scrollHeight;
}

function removeTyping() { const el = document.getElementById('typingInd'); if (el) el.remove(); }

function botReply(text) {
  showTyping();
  setTimeout(() => {
    removeTyping();
    const intent = matchIntent(text);
    if (intent && botRes[intent]) {
      addMsg(botRes[intent], 'bot');
    } else {
      addMsg(
        "Thank you for your enquiry. Your question falls outside the scope of what I am able to address through this channel. I would kindly recommend reaching out to our pharmacist directly — they will be best positioned to assist you with accurate and personalised guidance.",
        'bot', true, text
      );
    }
  }, 900 + Math.random() * 500);
}

function sendChatMessage() {
  const inp = document.getElementById('chatInput');
  const text = inp.value.trim();
  if (!text) return;
  addMsg(text, 'user');
  inp.value = '';
  inp.style.height = 'auto';
  botReply(text);
}

function handleChatKey(e) {
  if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendChatMessage(); }
}
